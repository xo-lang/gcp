// served: the body the fake answers the next request with
var served = ""

type fake struct{}

func (fake) RoundTrip(r *http.Request) (*http.Response, error) {
	var b []byte
	if r.Body != nil {
		b, _ = io.ReadAll(r.Body)
	}
	var hs []string
	for k, v := range r.Header {
		lk := strings.ToLower(k)
		if lk == "user-agent" || lk == "x-goog-api-client" || lk == "content-length" || lk == "accept-encoding" {
			continue
		}
		hs = append(hs, lk+"="+strings.Join(v, ","))
	}
	sort.Strings(hs)
	fmt.Printf("REQ %s %s\n  H %s\n  B %s\n  SERVED 200 %s\n", r.Method, r.URL.String(), strings.Join(hs, " | "), b, served)
	h := http.Header{}
	h.Set("Content-Type", "application/json")
	return &http.Response{StatusCode: 200, Status: "200 OK", Header: h, Body: io.NopCloser(strings.NewReader(served)), Request: r}, nil
}

func body[T any](s string) *T {
	v := new(T)
	if err := json.Unmarshal([]byte(s), v); err != nil {
		panic(err)
	}
	return v
}

func errLine(err error) {
	fmt.Printf("OUT: error: %s\n", strings.ReplaceAll(err.Error(), "\n", "\n| "))
}

func out(v any, err error) {
	if err != nil {
		errLine(err)
		return
	}
	b, e := json.Marshal(v)
	if e != nil {
		fmt.Printf("OUT: marshal error: %s\n", e)
		return
	}
	fmt.Printf("OUT: %s\n", b)
}

func outErr(err error) {
	if err != nil {
		errLine(err)
		return
	}
	fmt.Println("OUT: ok")
}
